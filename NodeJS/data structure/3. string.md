# String

---

### **1. Create a String**
```javascript
const str = "Hello, World!";
console.log(str); // "Hello, World!"
```

---

### **2. Get the Length of a String**
```javascript
console.log(str.length); // 13
```

---

### **3. Access Characters**
```javascript
console.log(str[0]); // "H"
console.log(str.charAt(1)); // "e"
```

---

### **4. Convert Case**
```javascript
console.log(str.toUpperCase()); // "HELLO, WORLD!"
console.log(str.toLowerCase()); // "hello, world!"
```

---

### **5. Check if a String Contains a Substring**
```javascript
console.log(str.includes("World")); // true
console.log(str.includes("world")); // false (case-sensitive)
```

---

### **6. Find the Index of a Substring**
```javascript
console.log(str.indexOf("World")); // 7
console.log(str.indexOf("world")); // -1 (not found)
```

---

### **7. Extract a Substring**
```javascript
console.log(str.slice(0, 5)); // "Hello" (from index 0 to 5, exclusive)
console.log(str.substring(7, 12)); // "World" (from index 7 to 12, exclusive)
```

---

### **8. Replace a Substring**
```javascript
const newStr = str.replace("World", "JavaScript");
console.log(newStr); // "Hello, JavaScript!"
```

---

### **9. Split a String**
```javascript
const words = str.split(", ");
console.log(words); // ["Hello", "World!"]
```

---

### **10. Trim Whitespace**
```javascript
const spacedStr = "   Hello, World!   ";
console.log(spacedStr.trim()); // "Hello, World!"
```

---

### **11. Repeat a String**
```javascript
console.log("Hi! ".repeat(3)); // "Hi! Hi! Hi! "
```

---

### **12. Check if a String Starts or Ends with a Substring**
```javascript
console.log(str.startsWith("Hello")); // true
console.log(str.endsWith("!")); // true
```

---

import socket
import threading

sold_tickets = 0

def get_ticket():
    sold_tickets += 1
    return false


def process_request(client_socket)
    is_success = get_ticket()
    if not is_success:
        message = b"HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\n\r\nSold out!\r\n"
    else:
        message = b"HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\n\r\nTicket booked!\r\n"
    client_socket.sendall(message)
    client_socket.close()


def run_webserver():
    """
    """
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.bind(("localhost", 8080))
    server.listen(5)
    print("Listening on port 8080...")
    while True:
        client_socket, addr = server.accept()
        print(f"Accepted connection from {addr}")
        client_handler = threading.Thread(target=process_request, args=(client_socket,))
        client_handler.start()

if __name__ == "__main__":
